cmake_minimum_required(VERSION 3.16)
project(autonomy_navigation)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/send_goal.py DESTINATION lib/${PROJECT_NAME} RENAME send_goal)
install(PROGRAMS scripts/velocity_guard.py DESTINATION lib/${PROJECT_NAME} RENAME velocity_guard)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(collision_monitor test/test_collision_monitor.py TIMEOUT 40)
endif()
ament_package()
