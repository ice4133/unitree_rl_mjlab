cmake_minimum_required(VERSION 3.16)
project(autonomy_perception)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(message_filters REQUIRED)

add_library(lidar_processing src/lidar_processing.cpp)
target_compile_features(lidar_processing PUBLIC cxx_std_17)
target_include_directories(lidar_processing PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(lidar_processing sensor_msgs geometry_msgs tf2 tf2_geometry_msgs)
add_executable(lidar_filter_node src/lidar_filter_node.cpp)
target_link_libraries(lidar_filter_node lidar_processing)
ament_target_dependencies(lidar_filter_node rclcpp tf2_ros message_filters)
install(TARGETS lidar_processing lidar_filter_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(lidar_processing_test test/lidar_processing_test.cpp)
  target_link_libraries(lidar_processing_test lidar_processing)
endif()
ament_package()
